% ================== FIG 1: SYSTEM ARCHITECTURE ==================
\begin{tikzpicture}[font=\scriptsize]
% ---- vehicle layer
\node[blk, fill=green!14, text width=40mm] (gt)
  {\textbf{GPS Tracker} (Concox GT06S)\\[-1pt]
   {\tiny \textbf{primary source} --- fully automatic}\\[-2pt]
   {\tiny fixed in the bus, no driver needed}};
\node[blk, fill=green!7, densely dashed, text width=28mm, anchor=north west]
  (ph) at ($(gt.north east)+(5.5mm,0)$)
  {\textbf{Driver Smartphone}\\[-1pt]
   {\tiny \textbf{automatic fallback}}\\[-2pt]
   {\tiny used only if needed}};

\node[zlab, anchor=west] at ($(vzone.north west)+(2.5mm,0)$) {Vehicle Layer};

% ---- backend layer
\node[blk, fill=blue!12, text width=24mm, below=10.5mm of gt.south west, anchor=north west] (trac)
  {\textbf{Telematics}\\[-1pt]\textbf{Gateway}\\[-1pt]{\tiny (Traccar) decodes}\\[-2pt]{\tiny tracker messages}};
\node[blk, fill=blue!8, text width=44mm, anchor=north west]
  (app) at ($(trac.north east)+(12mm,0)$)
  {\textbf{Application Server} {\tiny(Node.js)}\\[-1pt]
   {\tiny Source Selection $\cdot$ Data Validation}\\[-2pt]
   {\tiny ETA Calculation $\cdot$ Trip Monitoring}\\[-2pt]
   {\tiny Real-Time Broadcast}};
\node[blk, fill=orange!14, text width=34mm, anchor=north]
  (pg) at ($(trac.south west)!0.28!(app.south east)+(0,-4mm)$)
  {\textbf{PostgreSQL Database}\\[-1pt]{\tiny all trips, routes \& users}};
\node[blk, fill=orange!14, text width=32mm, anchor=north]
  (rd) at ($(trac.south west)!0.78!(app.south east)+(0,-4mm)$)
  {\textbf{Redis} (cache \& pub/sub)\\[-1pt]{\tiny instant update distribution}};
\node[zone, inner sep=2.8mm, fit=(trac)(app)(pg)(rd)] (bzone) {};
\node[zlab, anchor=west] at ($(bzone.north west)+(2.5mm,0)$) {Backend Server};

% ---- client layer
\node[blk, fill=black!6, text width=36mm, anchor=north west] (pass)
  at ($(bzone.south west)+(2.6mm,-9.5mm)$)
  {\textbf{Passenger Mobile App}\\[-1pt]
   {\tiny live map $\cdot$ arrival times $\cdot$ alerts}\\[-2pt]
   {\tiny bilingual English/Bangla}};
\node[blk, fill=black!6, text width=34mm, anchor=north east] (adm)
  at ($(bzone.south east)+(-2.6mm,-9.5mm)$)
  {\textbf{Admin Dashboard}\\[-1pt]
   {\tiny routes $\cdot$ schedules $\cdot$ fleet}\\[-2pt]
   {\tiny live operations map}};
\node[zone, inner sep=2.6mm, fit=(pass)(adm)] (czone) {};
\node[zlab] at (czone.north) {Client Layer};

% ---- connections

  node[lab, pos=0.45, anchor=west, xshift=0.8mm] {REST / HTTPS\\(fallback)};
\draw[arr] ($(trac.east)+(0,3.2mm)$) -- ($(app.west)+(0,3.2mm)$)
  node[lab, midway, anchor=south, yshift=0.3mm] {WebSocket\\($\approx$1\,s)};
\draw[bbarr] ($(trac.east)+(0,-3.6mm)$) -- ($(app.west)+(0,-3.6mm)$)
  node[lab, midway, anchor=north, yshift=-0.3mm] {REST poll\\(backup)};
\draw[darr, rounded corners=1.5pt] ($(app.south)+(-16mm,0)$) -- ++(0,-1.9mm) -| (pg.north);

\end{tikzpicture}
% ================== FIG 2: REAL-TIME DATA FLOW ==================
\begin{tikzpicture}[font=\scriptsize,
  stp/.style={blk, fill=blue!8, text width=61mm, inner sep=2.3pt},
  ph/.style={font=\tiny\bfseries\scshape, text=black!55, rotate=90, anchor=south}]
\node[stp] (s1)
  {\cnum{1} \textbf{GPS fix captured on the bus}\\[-1pt]
   {\tiny captured every 5\,s while the bus is moving}};
\node[stp, below=1.7mm of s1] (s2)
  {\cnum{2} \textbf{Transmitted and decoded}\\[-1pt]
   {\tiny sent over the cellular network; forwarded in $\approx$1\,s}};
\node[stp, below=3.4mm of s2] (s3)
  {\cnum{3} \textbf{Best source selected}\\[-1pt]
   {\tiny tracker preferred --- the phone is used only when the tracker is silent}};
\node[stp, below=1.7mm of s3] (s4)
  {\cnum{4} \textbf{Faulty readings rejected}\\[-1pt]
   {\tiny inaccurate positions, impossible jumps and GPS jitter are filtered out}};
\node[stp, below=1.7mm of s4] (s5)
  {\cnum{5} \textbf{Arrival times calculated}\\[-1pt]
   {\tiny remaining distance $\div$ recent speed, for every stop ahead}};
\node[stp, below=3.4mm of s5] (s6)
  {\cnum{6} \textbf{Stored and published}\\[-1pt]
   {\tiny saved in PostgreSQL; handed to Redis for instant distribution}};
\node[stp, below=1.7mm of s6] (s7)
  {\cnum{7} \textbf{Pushed to every subscribed phone}\\[-1pt]
   {\tiny WebSocket delivers the position and new arrival times per trip}};
\node[blk, fill=green!14, text width=61mm, inner sep=2.6pt, below=2.6mm of s7] (res)
  {\textbf{Passenger's map moves; arrival times refresh}\\[-1pt]
   {\tiny median delay, GPS fix to server: \textbf{5.9\,s} (measured in the pilot)}};
\foreach \a/\b in {s1/s2, s2/s3, s3/s4, s4/s5, s5/s6, s6/s7, s7/res}{
  \draw[-{Latex[length=1.5mm,width=1.3mm]}, semithick, draw=black!75] (\a.south) -- (\b.north);}
% phase brackets (left)
\foreach \top/\bot/\name in {s1/s2/Acquisition, s3/s5/Processing, s6/s7/Delivery}{
  \draw[black!55, line width=0.7pt]
    ($(\top.north west)+(-2.2mm,0)$) -- ($(\top.north west)+(-3.4mm,0)$) --
    ($(\bot.south west)+(-3.4mm,0)$) -- ($(\bot.south west)+(-2.2mm,0)$);}
\node[ph] at ($(s1.north west)!0.5!(s2.south west)+(-5.6mm,0)$) {Acquisition};
\node[ph] at ($(s3.north west)!0.5!(s5.south west)+(-5.6mm,0)$) {Processing};
\node[ph] at ($(s6.north west)!0.5!(s7.south west)+(-5.6mm,0)$) {Delivery};
% timeline (right)
\draw[black!55, line width=0.7pt]
  ($(s1.north east)+(3.2mm,0)$) -- ($(res.south east)+(3.2mm,0)$);
\foreach \n/\t in {s1/{t $=$ 0\,s}, s2/{$\approx$1\,s}, res/{5.9\,s median}}{
  \fill[black!60] ($(\n.east)+(3.2mm,0)$) circle (0.55mm);
  \node[font=\tiny, anchor=west, text=black!60] at ($(\n.east)+(4.4mm,0)$) {\t};}
\end{tikzpicture}
